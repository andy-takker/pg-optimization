# read.sql  — точечная выборка по PK
\set r random(1, 53000000)
SELECT id, fkRide, fio, contact, fkSeat FROM book.tickets WHERE id = :r;

# write.sql — INSERT с автокоммитом (генерация fio/contact)
INSERT INTO book.tickets (fkRide, fio, contact, fkSeat)
VALUES (ceil(random()*100),
  (array(SELECT fam FROM book.fam))[ceil(random()*110)]::text || ' ' ||
  (array(SELECT nam FROM book.nam))[ceil(random()*110)]::text,
  ('{"phone":"+7' || (1000000000::bigint + floor(random()*9000000000)::bigint)::text || '"}')::jsonb,
  ceil(random()*100));

# report.sql — тяжёлый отчёт по поездкам (раздельные агрегаты, наследие hw09)
WITH all_place AS (SELECT count(s.id) AS all_place, s.fkbus FROM book.seat s GROUP BY s.fkbus),
     order_place AS (SELECT count(t.id) AS order_place, t.fkride FROM book.tickets t GROUP BY t.fkride)
SELECT r.id, r.startdate AS depart_date, bs.city||', '||bs.name AS busstation, op.order_place, ap.all_place
FROM book.ride r JOIN book.schedule s ON r.fkschedule=s.id JOIN book.busroute br ON s.fkroute=br.id
JOIN book.busstation bs ON br.fkbusstationfrom=bs.id JOIN order_place op ON op.fkride=r.id
JOIN all_place ap ON r.fkbus=ap.fkbus ORDER BY r.startdate LIMIT 10;
